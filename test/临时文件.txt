// 获取Cookie函数
function getCookie(name) {
    let cookieValue = null;
    if (document.cookie && document.cookie !== '') {
        const cookies = document.cookie.split(';');
        for (let i = 0; i < cookies.length; i++) {
            const cookie = cookies[i].trim();
            if (cookie.substring(0, name.length + 1) === (name + '=')) {
                cookieValue = decodeURIComponent(cookie.substring(name.length + 1));
                break;
            }
        }
    }
    return cookieValue;
}

// 计算文字颜色函数 - 根据背景色计算最佳文字颜色（黑或白）
function getContrastYIQ(hexcolor) {
    // 如果没有提供颜色或格式不正确，返回默认黑色
    if (!hexcolor || !hexcolor.startsWith('#') || hexcolor.length !== 7) {
        return '#000000';
    }
    
    // 去掉#号并转换为RGB
    hexcolor = hexcolor.substring(1);
    const r = parseInt(hexcolor.substr(0, 2), 16);
    const g = parseInt(hexcolor.substr(2, 2), 16);
    const b = parseInt(hexcolor.substr(4, 2), 16);
    
    // 计算YIQ值 - 用于确定颜色的亮度
    const yiq = ((r * 299) + (g * 587) + (b * 114)) / 1000;
    
    // 如果背景色较亮返回黑色文字，否则返回白色文字
    return (yiq >= 128) ? '#000000' : '#ffffff';
}

    // 格式化日期时间为输入框格式
    function formatDateTimeForInput(date) {
        const d = new Date(date);
        
        // 获取小时和分钟
        let hours = d.getHours();
        let minutes = d.getMinutes();
        
        // 确保小时在6-24范围内
        hours = Math.max(6, Math.min(24, hours));
        
        // 将分钟调整到最近的15分钟间隔
        minutes = Math.round(minutes / 15) * 15;
        
        // 处理特殊情况：24点只能是24:00
        if (hours === 24 && minutes > 0) {
            hours = 24;
            minutes = 0;
        }
        
        // 格式化输出
        const hoursStr = String(hours).padStart(2, '0');
        const minutesStr = String(minutes).padStart(2, '0');
        
        return `${hoursStr}:${minutesStr}`;
    }
    
    // 设置时间选择器的值
    function setTimeSelectValue(selectId, timeValue) {
        const select = document.getElementById(selectId);
        if (select) {
            // 如果值不存在，找到最近的有效时间
            if (!Array.from(select.options).some(option => option.value === timeValue)) {
                const [hours, minutes] = timeValue.split(':').map(Number);
                const totalMinutes = hours * 60 + minutes;
                
                // 找到最近的可用选项
                let closestOption = null;
                let minDiff = Infinity;
                
                Array.from(select.options).forEach(option => {
                    if (option.value) {  // 跳过空选项
                        const [optHours, optMinutes] = option.value.split(':').map(Number);
                        const optTotalMinutes = optHours * 60 + optMinutes;
                        const diff = Math.abs(optTotalMinutes - totalMinutes);
                        
                        if (diff < minDiff) {
                            minDiff = diff;
                            closestOption = option.value;
                        }
                    }
                });
                
                timeValue = closestOption;
            }
            
            select.value = timeValue;
        }
    }

            year: 'numeric', 
            month: 'long', 
            day: 'numeric',
            hour: '2-digit',
            minute: '2-digit',
            hour12: false
        };
        
        // 如果是同一天，只显示一次日期
        if (startDate.toDateString() === endDate.toDateString()) {
            return `${startDate.toLocaleDateString('zh-CN', options)} - ${endDate.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit', hour12: false })}`;
        } else {
            return `${startDate.toLocaleDateString('zh-CN', options)} - ${endDate.toLocaleDateString('zh-CN', options)}`;
        }
    }

    // 保存任务
    function saveTask() {
        // 获取表单数据
        const taskId = document.getElementById('taskId').value;
        const title = document.getElementById('title').value;
        const subject = document.getElementById('subject').value;
        const chapter = document.getElementById('chapter').value;
        const startDateTime = document.getElementById('startDateTime').value;
        const endDateTime = document.getElementById('endDateTime').value;
        const description = document.getElementById('description').value;
        const focusLevel = document.getElementById('focusLevel').value;
        const energyLevel = document.getElementById('energyLevel').value;
        const repeatType = document.getElementById('repeatType').value;
        
        // 验证表单
        if (!title || !subject || !chapter || !startDateTime || !endDateTime) {
            alert('请填写所有必填字段');
            return;
        }
        
        // 构建任务数据
        const taskData = {
            title,
            subject,
            chapter,
            start_datetime: new Date(startDateTime).toISOString(),
            end_datetime: new Date(endDateTime).toISOString(),
            description,
            focus_level: focusLevel,
            energy_level: energyLevel,
            repeat_type: repeatType
        };
        
        // 如果有重复设置
        if (repeatType !== 'none') {
            const repeatEnds = document.getElementById('repeatEnds').value;
            if (repeatEnds) {
                taskData.repeat_ends = new Date(repeatEnds + 'T23:59:59').toISOString();
            }
            
            // 如果是每周重复或自定义，获取选中的日期
            if (repeatType === 'weekly' || repeatType === 'custom') {
                const repeatDays = [];
                for (let i = 0; i < 7; i++) {
                    if (document.getElementById(`day${i}`).checked) {
                        repeatDays.push(i);
                    }
                }
                taskData.repeat_days = repeatDays;
            }
        }
        
        // 确定是创建还是更新
        const method = taskId ? 'PUT' : 'POST';
        const url = taskId ? `/planner/api/tasks/${taskId}/` : '/planner/api/tasks/';
        
        // 发送请求
        fetch(url, {
            method,
            headers: {
                'Content-Type': 'application/json',
                'X-CSRFToken': csrftoken
            },
            body: JSON.stringify(taskData)
        })
        .then(response => {
            if (!response.ok) {
                return response.json().then(err => {
                    throw new Error(JSON.stringify(err));
                });
            }
            return response.json();
        })
        .then(data => {
            taskFormModal.hide();
            calendar.refetchEvents(); // 刷新日历
        })
        .catch(error => {
            console.error('Error saving task:', error);
            try {
                const errorData = JSON.parse(error.message);
                let errorMessage = '保存失败：\n';
                for (const key in errorData) {
                    errorMessage += `${key}: ${errorData[key]}\n`;
                }
                alert(errorMessage);
            } catch (e) {
                alert('保存任务失败，请检查表单数据');
            }
        });
    }

    // 根据背景色计算对比度最佳的文字颜色
    function getContrastYIQ(hexcolor) {
        // 移除 # 号（如果有）
        hexcolor = hexcolor.replace('#', '');
        
        // 转换为 RGB
        const r = parseInt(hexcolor.substr(0,2), 16);
        const g = parseInt(hexcolor.substr(2,2), 16);
        const b = parseInt(hexcolor.substr(4,2), 16);
        
        // 计算 YIQ 值
        const yiq = ((r * 299) + (g * 587) + (b * 114)) / 1000;
        
        // 根据 YIQ 值返回黑色或白色
        return (yiq >= 128) ? '#000000' : '#ffffff';
    }

    // 获取CSRF Token的辅助函数
    function getCookie(name) {
        let cookieValue = null;
        if (document.cookie && document.cookie !== '') {
            const cookies = document.cookie.split(';');
            for (let i = 0; i < cookies.length; i++) {
                const cookie = cookies[i].trim();
                if (cookie.substring(0, name.length + 1) === (name + '=')) {
                    cookieValue = decodeURIComponent(cookie.substring(name.length + 1));
                    break;
                }
            }
        }
        return cookieValue;
    }
